import { useGSAP } from "@gsap/react";

  gsap.registerPlugin(ScrollTrigger, useGSAP);
}

function StarRating({ rating }: { rating: number }) {
  return (
    <div className="flex items-center gap-0.5">
      {Array.from({ length: rating }).map((_, i) => (
        <Star key={i} size={14} weight="fill" className="text-gold" />
      ))}
    </div>
  );
}

function TestimonialCard({ testimonial }: { testimonial: Testimonial }) {
  return (
    <article className="flex flex-col justify-between h-full bg-dark-raised border border-border-dark rounded-2xl p-6 hover:border-gold/25 hover:shadow-[0_8px_32px_rgba(0,0,0,0.4)] transition-all duration-300">
      {/* Rating */}
      <div className="flex flex-col gap-4">
        <StarRating rating={testimonial.rating} />

        {/* Quote */}
        <blockquote className="text-[15px] text-white leading-relaxed">
          &#8220;{testimonial.quote}&#8221;
        </blockquote>
      </div>

      {/* Author */}
      <div className="flex items-center gap-3 mt-6 pt-5 border-t border-border-dark">
        <img
          src={testimonial.avatar}
          alt={`${testimonial.name} photo`}
          className="w-10 h-10 rounded-full object-cover border border-border-dark"
          width={40}
          height={40}
        />
        <div>
          <p className="text-[14px] font-semibold text-white">{testimonial.name}</p>
          <p className="text-[12px] text-text-dark-secondary">
            {testimonial.role} - {testimonial.location}
          </p>
        </div>
      </div>
    </article>
  );
}

export default function Testimonials() {
  const containerRef = useRef<HTMLElement>(null);
  
  const [prefersReducedMotion, setPrefersReducedMotion] = useState(true);

  useEffect(() => {
    setPrefersReducedMotion(window.matchMedia("(prefers-reduced-motion: reduce)").matches);
  }, []);

  const autoplay = Autoplay({ delay: 4500, stopOnInteraction: true });

  const [emblaRef, emblaApi] = useEmblaCarousel(
    { loop: true, align: "start" },
    prefersReducedMotion ? [] : [autoplay]
  );
  const [selectedIndex, setSelectedIndex] = useState(0);

  const scrollPrev = useCallback(() => emblaApi?.scrollPrev(), [emblaApi]);
  const scrollNext = useCallback(() => emblaApi?.scrollNext(), [emblaApi]);

  useEffect(() => {
    if (!emblaApi) return;
    const update = () => setSelectedIndex(emblaApi.selectedScrollSnap());
    emblaApi.on("select", update);
    return () => { emblaApi.off("select", update); };
  }, [emblaApi]);

  useGSAP(() => {
    if (prefersReducedMotion) return;

    gsap.from(".anim-header", {
      y: 24,
      opacity: 0,
      duration: 0.6,
      ease: "power3.out",
      scrollTrigger: {
        trigger: ".anim-header",
        start: "top 80%",
      }
    });

    gsap.from(".anim-controls", {
      x: 20,
      opacity: 0,
      duration: 0.6,
      delay: 0.1,
      ease: "power3.out",
      scrollTrigger: {
        trigger: ".anim-header",
        start: "top 80%",
      }
    });

    gsap.from(".anim-stats", {
      y: 16,
      opacity: 0,
      duration: 0.6,
      delay: 0.2,
      ease: "power3.out",
      scrollTrigger: {
        trigger: ".anim-stats",
        start: "top 90%",
      }
    });
  }, { scope: containerRef, dependencies: [prefersReducedMotion] });

  return (
    <section ref={containerRef} className="py-24 md:py-32 bg-dark-base relative overflow-hidden" id="testimonials">
      {/* Background accent */}
      <div
        className="absolute bottom-0 left-0 w-[500px] h-[400px] pointer-events-none"
        style={{ background: "radial-gradient(ellipse at bottom left, rgba(223,158,4,0.04) 0%, transparent 70%)" }}
      />

      <div className="max-w-[1400px] mx-auto px-5 md:px-10">
        {/* Header */}
        <div className="flex flex-col md:flex-row md:items-end justify-between gap-6 mb-10">
          <div className="anim-header">
            <h2 className="font-display text-headline-lg text-white leading-tight">
              What Clients{" "}
              <span className="text-gold">Say</span>
            </h2>
            <p className="text-text-dark-secondary text-body-default mt-2 max-w-md">
              Real words from real Kolkata home-buyers and sellers.
            </p>
          </div>

          <div className="anim-controls flex items-center gap-3 shrink-0">
            <button
              onClick={scrollPrev}
              className="w-10 h-10 rounded-full border border-border-dark bg-dark-raised flex items-center justify-center text-text-dark-secondary hover:text-gold hover:border-gold/40 transition-all duration-200 active:scale-95"
              aria-label="Previous testimonial"
            >
              <ArrowLeft size={16} />
            </button>
            <button
              onClick={scrollNext}
              className="w-10 h-10 rounded-full border border-border-dark bg-dark-raised flex items-center justify-center text-text-dark-secondary hover:text-gold hover:border-gold/40 transition-all duration-200 active:scale-95"
              aria-label="Next testimonial"
            >
              <ArrowRight size={16} />
            </button>
          </div>
        </div>

        {/* Embla carousel */}
        <div className="embla overflow-hidden" ref={emblaRef}>
          <div className="embla__container flex gap-5">
            {TESTIMONIALS.map((t) => (
              <div
                key={t.id}
                className="embla__slide shrink-0"
                style={{ flex: "0 0 min(380px, 90vw)" }}
              >
                <TestimonialCard testimonial={t} />
              </div>
            ))}
          </div>
        </div>

        {/* Dots */}
        <div className="flex justify-center gap-2 mt-8" role="tablist" aria-label="Testimonials navigation">
          {TESTIMONIALS.map((_, i) => (
            <button
              key={i}
              role="tab"
              aria-selected={i === selectedIndex}
              aria-label={`Go to testimonial ${i + 1}`}
              onClick={() => emblaApi?.scrollTo(i)}
              className={`h-1.5 rounded-full transition-all duration-300 cursor-pointer ${
                i === selectedIndex ? "bg-gold w-6" : "bg-border-dark w-1.5 hover:bg-text-dark-muted"
              }`}
            />
          ))}
        </div>

        {/* Trust strip */}
        <div className="anim-stats mt-12 flex flex-wrap justify-center gap-8 md:gap-12">
          {[
            { value: "1,200+", label: "Happy Clients" },
            { value: "4.9",    label: "Average Rating" },
            { value: "98%",    label: "Referral Rate" },
          ].map((stat, i) => (
            <div key={i} className="text-center">
              <div className="font-display text-[28px] font-bold text-gold">{stat.value}</div>
              <div className="text-[12px] text-text-dark-secondary mt-0.5">{stat.label}</div>
            </div>
          ))}
        </div>
      </div>
    </section>
  );
}
